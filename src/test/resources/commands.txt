# CucumberTestSuite (Java 17+, Maven; defaults to @CartPrice)
mvn clean verify

# Override the default Cucumber tag expression
mvn clean verify "-Dcucumber.filter.tags=(@Smoke or @Regression) and not @manual"

# Headless Chrome
mvn clean verify -Dheadless.mode=true

# Remote Selenium Grid
mvn clean verify -Denvironment=grid -Dwebdriver.remote.url=http://localhost:4444 -Dheadless.mode=true
